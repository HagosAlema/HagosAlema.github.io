NFT Royalty Standard (EIP-2981)

UnilapseNFT.sol 
contract UnilapseNFT is ERC721Enumerable, ERC2981 {
    function mintWithRoyalty(
        address to, string memory tokenURI,
        address receiver, uint96 feeNumerator
    ) external onlyOwner {
        uint256 tokenId = _nextTokenId();
        _safeMint(to, tokenId);
        _setTokenRoyalty(tokenId, receiver, feeNumerator);
    }
}
